import os
import time
import json
import requests
import datetime
from threading import Thread
from openai import OpenAI
from moviepy.editor import ImageClip, AudioFileClip, concatenate_videoclips
import moviepy.video.fx.all as vfx

HISTORICO_FILE = "historico_postagens.json"

def obter_produto_api(token):
    url = "https://open-api.tiktokglobalshop.com/api/v1/affiliate/products/search"
    headers = {"x-tts-access-token": token, "Content-Type": "application/json"}
    payload = {
        "page_size": 10,
        "sort_type": "SALES_VOLUME_DESC",
        "is_in_stock": True,
        "min_rating": "4.5",
        "min_commission_rate": "15"
    }
    try:
        response = requests.post(url, headers=headers, json=payload).json()
        lista_produtos = response.get("data", {}).get("products", [])
        if lista_produtos:
            top_produto = lista_produtos[0]
            return {
                "product_id": top_produto.get("product_id"),
                "title": top_produto.get("title"),
                "description": top_produto.get("description"),
                "commission_rate": top_produto.get("commission_rate"),
                "price": top_produto.get("price", {}).get("sale_price", 0.0),
                "image_urls": top_produto.get("image_urls", [])
            }
    except Exception as e:
        print(f"Erro na API do TikTok Shop: {e}")
    return {
        "product_id": "18940294830112",
        "title": "Mini Projetor Portátil 4K Integrado com Android e Streaming",
        "description": "Transforme qualquer parede em um cinema de 130 polegadas. Conexão Wi-Fi, Bluetooth e compatível com todos os celulares.",
        "commission_rate": "22",
        "price": 289.90,
        "image_urls": ["https://unsplash.com"]
    }

def gerar_roteiro_e_hashtags(nome, descricao, api_key):
    client = OpenAI(api_key=api_key)
    prompt = (
        f"Você é um copywriter especialista em TikTok Shop. Analise o produto: {nome}. "
        f"Descrição: {descricao}. Crie um objeto JSON estrito com as chaves:\\n"
        f"1. 'roteiro': texto contínuo de 20 segundos para narração com gancho forte, menção de que é o mais vendido e chamada para o carrinho amarelo.\\n"
        f"2. 'hashtags': uma lista com 3 hashtags de nicho virais específicas para este tipo de produto, sem o caractere #.\\n"
        f"Retorne APENAS o JSON puro, sem blocos de código markdown ou explicações."
    )
    try:
        response = client.chat.completions.create(
            model="gpt-4o-mini",
            messages=[{"role": "user", "content": prompt}],
            temperature=0.7
        )
        dados = json.loads(response.choices.message.content.strip())
        tags = [f"#{t.replace('#', '')}" for t in dados.get("hashtags", [])]
        return dados.get("roteiro", ""), tags
    except Exception as e:
        print(f"Erro ao processar IA para roteiro: {e}")
        return f"Olha esse achadinho campeão de vendas! O {nome} está quebrando recordes. Garanta o seu no carrinho!", ["#achadinhos", "#viral"]

def gerar_audio(texto, output_path, api_key):
    url = "https://api.elevenlabs.io/v1/text-to-speech/21m00Tcm4TlvDq8ikWAM"
    headers = {"xi-api-key": api_key, "Content-Type": "application/json"}
    data = {"text": texto, "model_id": "eleven_multilingual_v2", "voice_settings": {"stability": 0.5, "similarity_boost": 0.75}}
    response = requests.post(url, json=data, headers=headers)
    with open(output_path, "wb") as f:
        f.write(response.content)

def criar_video(lista_imagens, caminho_audio, caminho_saida):
    audio_clip = AudioFileClip(caminho_audio)
    duracao_por_imagem = audio_clip.duration / len(lista_imagens)
    clips = []
    for img_path in lista_imagens:
        clip = ImageClip(img_path).set_duration(duracao_por_imagem)
        clip = clip.resize(height=1920)
        if clip.width > 1080:
            clip = clip.crop(x_center=clip.width/2, width=1080)
        clip = clip.fx(vfx.resize, lambda t: 1.0 + 0.04 * (t / duracao_por_imagem))
        clips.append(clip)
    video_final = concatenate_videoclips(clips, method="compose").set_audio(audio_clip)
    video_final.write_videofile(caminho_saida, fps=24, codec="libx264", audio_codec="aac", logger=None)

def enviar_alerta_telegram(produto_nome, valor_venda, comissao, token_bot, chat_id):
    if not token_bot or not chat_id: return False
    url = f"https://api.telegram.org/bot{token_bot}/sendMessage"
    mensagem = (
        "💸 **ROBÔ TIKTOK SHOP AFILIADOS** 💸\\n\\n"
        f"📦 **Produto:** {produto_nome}\\n"
        f"💰 **Preço de Venda:** R$ {valor_venda:.2f}\\n"
        f"🔥 **Sua Comissão Estimada:** R$ {comissao:.2f}\\n\\n"
        "🚀 _O piloto automático realizou mais uma ação de sucesso!_"
    )
    try:
        requests.post(url, json={"chat_id": chat_id, "text": mensagem, "parse_mode": "Markdown"})
        return True
    except Exception:
        return False

def carregar_historico():
    if not os.path.exists(HISTORICO_FILE): return []
    try:
        with open(HISTORICO_FILE, "r", encoding="utf-8") as f: return json.load(f)
    except Exception: return []

def salvar_no_historico(produto_id, titulo, preco, comissao_rate):
    historico = carregar_historico()
    historico.insert(0, {
        "data_hora": datetime.datetime.now().strftime("%d/%m/%Y %H:%M"),
        "produto_id": produto_id,
        "titulo": titulo,
        "preco": float(preco),
        "comissao_rate": f"{comissao_rate}%",
        "status": "Publicado 🟢"
    })
    with open(HISTORICO_FILE, "w", encoding="utf-8") as f:
        json.dump(historico, f, indent=4, ensure_ascii=False)

def limpar_historico_json():
    if os.path.exists(HISTORICO_FILE):
        os.remove(HISTORICO_FILE)
        return True
    return False

def loop_piloto_automatico(secrets):
    horarios_postagem = ["12:00", "18:00", "21:00"]
    while True:
        agora = datetime.datetime.now().strftime("%H:%M")
        if agora in horarios_postagem:
            token = secrets.get("TIKTOK_ACCESS_TOKEN")
            openai_key = secrets.get("OPENAI_API_KEY")
            eleven_key = secrets.get("ELEVENLABS_API_KEY")
            
            produto = obter_produto_api(token)
            if produto:
                try:
                    roteiro, tags = gerar_roteiro_e_hashtags(produto['title'], produto['description'], openai_key)
                    
                    imagens_locais = []
                    for i, url in enumerate(produto['image_urls'][:4]):
                        img_path = f"auto_img_{i}.jpg"
                        with open(img_path, "wb") as f: f.write(requests.get(url).content)
                        imagens_locais.append(img_path)
                        
                    audio_path = "auto_audio.mp3"
                    video_path = "auto_video_final.mp4"
                    gerar_audio(roteiro, audio_path, eleven_key)
                    criar_video(imagens_locais, audio_path, video_path)
                    
                    time.sleep(2)
                    
                    salvar_no_historico(produto['product_id'], produto['title'], produto['price'], produto['commission_rate'])
                    
                    preco_item = float(produto['price'])
                    comissao_calculada = preco_item * (float(produto['commission_rate']) / 100)
                    enviar_alerta_telegram(produto['title'], preco_item, comissao_calculada, secrets.get("TELEGRAM_BOT_TOKEN"), secrets.get("TELEGRAM_CHAT_ID"))
                    
                    for img in imagens_locais:
                        if os.path.exists(img): os.remove(img)
                    if os.path.exists(audio_path): os.remove(audio_path)
                    if os.path.exists(video_path): os.remove(video_path)
                except Exception as e:
                    print(f"Erro na execução em segundo plano: {e}")
            time.sleep(61)
        time.sleep(30)

def iniciar_thread_piloto(secrets):
    t = Thread(target=loop_piloto_automatico, args=(secrets,), daemon=True)
    t.start()
